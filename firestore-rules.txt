هذا النص خاص بحماية قاعدة البيانات. الصقه في:
Firebase Console → مشروعك → Firestore Database → تبويب "Rules" → استبدل كل المحتوى بهذا → اضغط "Publish"

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // الجميع يقدر يشوف المنتجات/الخدمات/طرق الدفع/الإعدادات (باش يشتغل المتجر للزوار)
    // لكن التعديل والحذف والإضافة مسموح فقط للأدمن المسجّل دخوله (Firebase Authentication)
    match /products/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    match /services/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    match /payments/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    match /settings/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
